class _SubmitEwasteScreenState extends State<SubmitEwasteScreen> {
  final _photoPicker = PhotoPickerService();
  final _locationService = LocationService();
  final _categoryController = TextEditingController();
  final _weightController = TextEditingController();

  // Generated once per submission attempt and reused across retries of
  // that SAME attempt, so a slow-network double-tap on Submit can't create
  // two separate submissions server-side.
  late final String _idempotencyKey;

  XFile? _photo;
  Uint8List? _photoBytes;
  bool _pickingPhoto = false;
  String? _photoError;

  bool _locating = false;
  String? _locationError;
  double? _lat;
  double? _lng;

  bool _submitting = false;
  String? _submitError;
  SubmissionCreateResult? _result;

  @override
  void initState() {
    super.initState();
    _idempotencyKey = IdempotencyKeyGenerator.generate();
  }

  @override
  void dispose() {
    _categoryController.dispose();
    _weightController.dispose();
    super.dispose();
  }

  Future<void> _pickPhoto(ImageSource source) async {
    setState(() {
      _pickingPhoto = true;
      _photoError = null;
    });
    try {
      final file = await _photoPicker.pickPhoto(source: source);
      if (file == null) {
        if (mounted) setState(() => _pickingPhoto = false);
        return;
      }
      final bytes = await file.readAsBytes();
      if (!mounted) return;
      setState(() {
        _photo = file;
        _photoBytes = bytes;
        _pickingPhoto = false;
      });
    } catch (_) {
      if (!mounted) return;
      setState(() {
        _photoError = 'Could not access the camera/gallery.';
        _pickingPhoto = false;
      });
    }
  }

  Future<void> _captureLocation() async {
    setState(() {
      _locating = true;
      _locationError = null;
    });
    try {
      final position = await _locationService.getCurrentLocation();
      if (!mounted) return;
      setState(() {
        _lat = position.latitude;
        _lng = position.longitude;
      });
    } on LocationException catch (e) {
      if (!mounted) return;
      setState(() => _locationError = e.message);
    } finally {
      if (mounted) setState(() => _locating = false);
    }
  }

  Future<void> _reviewAndSubmit() async {
    if (_photo == null) {
      setState(() => _submitError = 'Please capture or select a photo first.');
      return;
    }
    if (_lat == null || _lng == null) {
      setState(() => _submitError = 'Please capture your location first.');
      return;
    }
    final category = _categoryController.text.trim();
    final weight = double.tryParse(_weightController.text.trim());
    if (category.isEmpty) {
      setState(() => _submitError = 'Please enter a material or category.');
      return;
    }
    if (weight == null || weight <= 0) {
      setState(
        () => _submitError =
            'Please enter an approximate weight greater than zero.',
      );
      return;
    }
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (context) => AlertDialog(
        title: const Text('Review pickup request'),
        content: Text(
          'Category: $category\nApproximate weight: ${weight.toStringAsFixed(2)} kg\n'
          'Pickup location: ${_lat!.toStringAsFixed(5)}, ${_lng!.toStringAsFixed(5)}',
        ),
        actions: [
          TextButton(
            onPressed: () => Navigator.pop(context, false),
            child: const Text('Edit'),
          ),
          ElevatedButton(
            onPressed: () => Navigator.pop(context, true),
            child: const Text('Submit request'),
          ),
        ],
      ),
    );
    if (confirmed != true || !mounted) return;
    await _submit(category: category, weight: weight);
  }

  Future<void> _submit({
    required String category,
    required double weight,
  }) async {
    setState(() {
      _submitting = true;
      _submitError = null;
    });
    final token = context.read<AuthController>().session?.accessToken;
    if (token == null) {
      setState(() {
        _submitting = false;
        _submitError = 'Your session has expired. Please log in again.';
      });
      return;
    }

    try {
      final result = await context.read<EwasteRepository>().createSubmission(
        accessToken: token,
        wardId: widget.wardId,
        photoFileId: PhotoPickerService.localPhotoReference(),
        homeLat: _lat!,
        homeLng: _lng!,
        category: category,
        approximateWeightKg: weight,
        idempotencyKey: _idempotencyKey,
      );
      if (!mounted) return;
      setState(() {
        _result = result;
        _submitting = false;
      });
    } on ApiException catch (e) {
      if (!mounted) return;
      setState(() {
        _submitError = e.message;
        _submitting = false;
      });
    }
  }

"""
AI-driven classification (architecture Section 9), a sibling to the
existing manual classify() in app/modules/ewaste/service.py — NOT a
replacement for it. Both write to the same append-only
ai_classifications collection and drive the same
state_machines/ewaste_states.py; this module owns only the AT_RRR ->
{CLASSIFIED, NEEDS_REVIEW} decision based on the classifier's result.

Money-equivalent safety principle applied here: never silently accept an
unreliable result. success=False OR confidence below the configured
threshold OR a malformed ClassificationResult all route to NEEDS_REVIEW,
exactly like an unresolved handover routes to FLAGGED_FOR_REVIEW instead
of silently "succeeding".
"""
from datetime import datetime, timezone
from uuid import uuid4

from app.core import audit, idempotency
from app.core.config import settings
from app.core.errors import bad_request, forbidden, not_found
from app.db.repositories import classifications_repo, submissions_repo
from app.modules.classification.classifier import ClassificationResult, ClassifierError, get_classifier
from app.modules.rrr_centres import service as rrr_centres_service
from app.state_machines.ewaste_states import transition

AI_CLASSIFY_IDEMPOTENCY_SCOPE = "ai_classify"


def _is_well_formed(result: ClassificationResult) -> bool:
    """Defends against a malformed classifier response (e.g. a future
    real provider returning a garbage payload) — this is deliberately
    checked here, not trusted from the classifier, since the classifier
    is a swappable, potentially-untrusted-input boundary."""
    if not result.success:
        return True  # a reported failure doesn't need category/confidence to be valid
    if not isinstance(result.predicted_category, str) or not result.predicted_category.strip():
        return False
    if not isinstance(result.confidence, (int, float)):
        return False
    if isinstance(result.confidence, bool):  # bool is technically an int subclass
        return False
    if not (0.0 <= float(result.confidence) <= 1.0):
        return False
    return True


async def classify_with_ai(*, current_user: dict, submission_id: str, idempotency_key: str) -> dict:
    cached = await idempotency.get_cached_response(current_user["user_id"], idempotency_key, AI_CLASSIFY_IDEMPOTENCY_SCOPE)
    if cached is not None:
        return cached

    rrr_profile = await rrr_centres_service.require_approved_rrr(current_user)

    submission = await submissions_repo.find_by_submission_id(submission_id)
    if submission is None:
        raise not_found("SUBMISSION_NOT_FOUND", "No submission exists with this submission_id.")
    if submission["ward_id"] != rrr_profile["ward_id"]:
        raise forbidden("RRR_OUTSIDE_ASSIGNED_WARD", "This submission is outside your assigned ward.")
    if submission["status"] != "AT_RRR":
        raise bad_request(
            "SUBMISSION_NOT_AT_RRR",
            f"Submission is in status {submission['status']}, not eligible for AI classification.",
        )

    classifier = get_classifier()
    try:
        result = await classifier.classify(submission["photo_file_id"])
        if not _is_well_formed(result):
            result = ClassificationResult(
                success=False,
                predicted_category=None,
                confidence=None,
                model_version=result.model_version or settings.ai_model_version,
                failure_reason="MALFORMED_CLASSIFIER_RESPONSE",
            )
    except ClassifierError as exc:
        result = ClassificationResult(
            success=False,
            predicted_category=None,
            confidence=None,
            model_version=settings.ai_model_version,
            failure_reason=str(exc) or "CLASSIFIER_ERROR",
        )
    except Exception as exc:  # never let a classifier bug crash the RRR workflow
        result = ClassificationResult(
            success=False,
            predicted_category=None,
            confidence=None,
            model_version=settings.ai_model_version,
            failure_reason=f"UNEXPECTED_CLASSIFIER_EXCEPTION: {type(exc).__name__}",
        )

    accepted = result.success and result.confidence is not None and result.confidence >= settings.classification_confidence_threshold
    now = datetime.now(timezone.utc)

    if accepted:
        new_status = transition("AT_RRR", "CLASSIFIED")
        await submissions_repo.update_status_and_category(submission_id, new_status, result.predicted_category)
    else:
        new_status = transition("AT_RRR", "NEEDS_REVIEW")
        await submissions_repo.update_status(submission_id, new_status)

    classification_id = str(uuid4())
    await classifications_repo.insert_classification(
        {
            "classification_id": classification_id,
            "item_id": submission_id,
            "predicted_category": result.predicted_category,
            "confidence": result.confidence,
            "model_version": result.model_version,
            "confirmed_category": result.predicted_category if accepted else None,
            "confirmed_by": None,  # AI-only acceptance has no human confirmer, by design
            "source": "AI",
            "status": "ACCEPTED" if accepted else ("FAILED" if not result.success else "LOW_CONFIDENCE"),
            "failure_reason": result.failure_reason,
            "created_at": now,
        }
    )

    await audit.record(
        actor_id=current_user["user_id"],
        actor_role=current_user["role"],
        action="AI_CLASSIFICATION_ATTEMPTED",
        entity_type="e_waste_submission",
        entity_id=submission_id,
        details={
            "classification_id": classification_id,
            "accepted": accepted,
            "new_status": new_status,
            "predicted_category": result.predicted_category,
            "confidence": result.confidence,
        },
    )

    response = {
        "submission_id": submission_id,
        "status": new_status,
        "category": result.predicted_category if accepted else None,
        "predicted_category": result.predicted_category,
        "confidence": result.confidence,
        "model_version": result.model_version,
        "source": "AI",
        "needs_review": not accepted,
        "failure_reason": result.failure_reason,
    }
    await idempotency.store_response(current_user["user_id"], idempotency_key, AI_CLASSIFY_IDEMPOTENCY_SCOPE, response)
    return response

from pydantic import BaseModel, ConfigDict


class AiClassifyResponse(BaseModel):
    # model_version isn't a Pydantic protected-namespace clash in intent —
    # it matches the field name already established across
    # ai_classifications/config.py/architecture docs — so the namespace
    # check is disabled here rather than renaming the field.
    model_config = ConfigDict(protected_namespaces=())

    submission_id: str
    status: str  # "CLASSIFIED" or "NEEDS_REVIEW"
    category: str | None  # the accepted category, only set when status == CLASSIFIED
    predicted_category: str | None  # whatever the classifier guessed, even if not accepted
    confidence: float | None
    model_version: str
    source: str = "AI"
    needs_review: bool
    failure_reason: str | None = None

from abc import ABC, abstractmethod
from dataclasses import dataclass, field
from typing import Any, Optional


@dataclass
class ClassificationResult:
    success: bool
    predicted_category: Optional[str] = None
    confidence: float = 0.0
    model_version: str = "unknown"
    failure_reason: Optional[str] = None
    metadata: dict[str, Any] = field(default_factory=dict)


class ClassifierError(Exception):
    """Raised when classification fails."""
    pass


class ClassifierService(ABC):

    @abstractmethod
    async def classify(self, photo_file_id: str) -> ClassificationResult:
        """Classify an image using the configured model."""
        raise NotImplementedError


def get_classifier() -> ClassifierService:
    """
    Return the YOLO classifier.

    The import is inside the function to prevent circular imports.
    """
    from .yolo_classifier import YOLOClassifier

    return YOLOClassifier()

from pathlib import Path
import asyncio

from ultralytics import YOLO

from .classifier import (
    ClassifierService,
    ClassificationResult,
)


class YOLOClassifier(ClassifierService):
    def __init__(self):
        model_path = (
            Path(__file__).resolve().parents[3]
            / "ai"
            / "detection"
            / "weights"
            / "best.pt"
        )

        self.model = YOLO(str(model_path))
        self.model_version = "YOLO-best.pt"

    def map_category(self, detected_label: str) -> str:
        label = detected_label.lower()

        if "battery" in label:
            return "batteries"

        if label in ["pcb"]:
            return "circuit-boards"

        if label in [
            "computer-keyboard",
            "computer-mouse",
            "router",
            "usb-flash-drive",
            "telephone-set",
            "tv-remote-control",
        ]:
            return "e-waste-small-appliances"

        return "e-waste-small-appliances"

    async def classify(self, photo_file_id: str) -> ClassificationResult:
        try:
            image_path = Path(photo_file_id)

            if not image_path.exists():
                return ClassificationResult(
                    success=False,
                    predicted_category=None,
                    confidence=0.0,
                    model_version=self.model_version,
                    failure_reason=f"Image not found: {photo_file_id}",
                    metadata={},
                )

            results = await asyncio.to_thread(
                self.model.predict,
                source=str(image_path),
                verbose=False,
            )

            result = results[0]

            if result.boxes is None or len(result.boxes) == 0:
                return ClassificationResult(
                    success=False,
                    predicted_category=None,
                    confidence=0.0,
                    model_version=self.model_version,
                    failure_reason="No object detected",
                    metadata={},
                )

            best_index = result.boxes.conf.argmax().item()

            confidence = float(result.boxes.conf[best_index])
            class_id = int(result.boxes.cls[best_index])
            detected_label = result.names[class_id]

            category = self.map_category(detected_label)

            return ClassificationResult(
                success=True,
                predicted_category=category,
                confidence=confidence,
                model_version=self.model_version,
                failure_reason=None,
                metadata={
                    "detected_label": detected_label,
                    "class_id": class_id,
                },
            )

        except Exception as error:
            return ClassificationResult(
                success=False,
                predicted_category=None,
                confidence=0.0,
                model_version=self.model_version,
                failure_reason=str(error),
                metadata={},
            )

from app.core.errors import bad_request


def predict_price(
    *,
    product_type: str,
    condition: str,
    quantity: int,
    total_weight_kg: float,
    location: str,
) -> float:
    normalized_product_type = product_type.strip()
    normalized_condition = condition.strip()
    normalized_location = location.strip()

    if not normalized_product_type:
        raise bad_request("PRODUCT_TYPE_REQUIRED", "Product type is required.")
    if not normalized_condition:
        raise bad_request("CONDITION_REQUIRED", "Condition is required.")
    if not normalized_location:
        raise bad_request("LOCATION_REQUIRED", "Location is required.")
    if quantity <= 0:
        raise bad_request("INVALID_QUANTITY", "Quantity must be greater than zero.")
    if total_weight_kg <= 0:
        raise bad_request("INVALID_TOTAL_WEIGHT", "Total weight must be greater than zero.")

    # Load the model only when pricing is requested, not during API startup.
    from ai.pricing.price_predictor import predict_collector_price

    return predict_collector_price(
        product_type=normalized_product_type,
        condition=normalized_condition,
        quantity=quantity,
        total_weight_kg=total_weight_kg,
        location=normalized_location,
    )

class PricePredictionRequest(BaseModel):
    product_type: str = Field(min_length=1, max_length=100)
